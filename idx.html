from flask import Flask, request, render_template_string import os app = Flask(__name__) UPLOAD_FOLDER = "uploads" os.makedirs(UPLOAD_FOLDER, exist_ok=True) HTML_PAGE = """ Upload Multiple Photos

Upload Multiple Images



""" @app.route("/", methods=["GET", "POST"]) def upload(): if request.method == "POST": files = request.files.getlist("photos") for file in files: if file.filename != "": file.save(os.path.join(UPLOAD_FOLDER, file.filename)) return "Photos uploaded successfully ✅" return render_template_string(HTML_PAGE) if __name__ == "__main__": app.run(debug=True)
